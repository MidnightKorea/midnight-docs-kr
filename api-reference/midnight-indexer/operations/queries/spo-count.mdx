export const Details = ({
  dataOpen,
  dataClose,
  children,
  startOpen = false
}) => {
  const [open, setOpen] = useState(startOpen);
  return (
    <details
      {...(open ? { open: true } : {})}
      className="details"
      style={{
        border: "none",

        background: "var(--ifm-background-color)"

      </summary>
      {open && children}

  );

### Type

#### [<code style={{ fontWeight: 'normal' }}><b>Int</b></code>](/midnight-indexer/types/scalars/int.mdx) <Badge class="badge badge--secondary " text="scalar"/>

`Int` 스칼라 타입은 부호 있는 정수 값을 나타냅니다. -(2^31)에서 2^31 - 1 사이의 값을 표현할 수 있습니다.
